    boxed title style={colback=hessoGrey, arc=3pt, boxrule=0pt}]

  \begin{tabularx}{\linewidth}{@{}p{3.5cm}X@{}}
    \textbf{Date de production} & #1 \\
    \textbf{Contexte d'origine} & #2 \\
    \textbf{Mobilisé comme preuve de} & #3 \\
    \textbf{Niveau de caviardage} & #4 \\
